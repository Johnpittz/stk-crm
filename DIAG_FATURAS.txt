-- DIAGNÓSTICO: Por que faturas não aparecem?
-- ============================================

-- 1. O cliente "teste" existe em clientes_reciee?
SELECT id, nome, cpf_cnpj, uc, estado, distribuidora, grupo
FROM clientes_reciee
WHERE cpf_cnpj ILIKE '%002388700%'
   OR nome ILIKE '%teste%';

-- 2. Tem faturas na tabela faturas_reciee?
SELECT COUNT(*) as total_faturas FROM faturas_reciee;

-- 3. Quais clientes têm faturas?
SELECT fr.cliente_id, cr.nome, cr.cpf_cnpj, COUNT(*) as qtd_faturas
FROM faturas_reciee fr
LEFT JOIN clientes_reciee cr ON fr.cliente_id = cr.id
GROUP BY fr.cliente_id, cr.nome, cr.cpf_cnpj;

-- 4. O unified_clients tem source_id = id do clientes_reciee?
SELECT uc.id, uc.source_id, uc.nome, uc.documento
FROM unified_clients uc
WHERE uc.source = 'reciee';
