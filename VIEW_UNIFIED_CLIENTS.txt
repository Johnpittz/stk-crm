-- VIEW: unified_clients com métricas
-- Colunas da tabela: id, source, source_id, nome, documento, telefone, email, uc, estado, distribuidora, grupo, classify_chatbot, created_at

CREATE OR REPLACE VIEW unified_clients_full AS
SELECT
  uc.*,
  CASE WHEN EXISTS (
    SELECT 1 FROM atendimentos a
    WHERE a.cliente_id::text = uc.id::text
       OR a.telefone_cliente = uc.telefone
  ) THEN true ELSE false END AS tem_atendimento,
  CASE WHEN EXISTS (
    SELECT 1 FROM chatbot_sessions cs
    WHERE cs.telefone = uc.telefone
  ) THEN true ELSE false END AS tem_chatbot,
  CASE WHEN EXISTS (
    SELECT 1 FROM faturas_reciee fr
    WHERE fr.cliente_id::text = uc.id::text
  ) THEN true ELSE false END AS tem_faturas_reciee,
  uc.created_at AS ultima_interacao
FROM unified_clients uc;
