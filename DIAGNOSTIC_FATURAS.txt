-- Diagnóstico: verificar se o CPF bate entre as tabelas
-- Rode no Supabase SQL Editor

-- 1. Ver CPF do cliente "teste" na tabela clientes
SELECT id, nome_razao_social, cnpj_cpf FROM clientes WHERE nome_razao_social ILIKE '%teste%';

-- 2. Ver CPF do cliente "teste" na tabela clientes_reciee  
SELECT id, nome, cpf_cnpj FROM clientes_reciee WHERE nome ILIKE '%teste%';

-- 3. Ver faturas do cliente "teste"
SELECT f.id, f.competencia, f.valor_total, f.cliente_id
FROM faturas_reciee f
JOIN clientes_reciee cr ON cr.id = f.cliente_id
WHERE cr.nome ILIKE '%teste%';

-- 4. Comparar formatos de CPF
SELECT 'clientes' as tabela, cnpj_cpf as cpf FROM clientes WHERE nome_razao_social ILIKE '%teste%'
UNION ALL
SELECT 'clientes_reciee' as tabela, cpf_cnpj as cpf FROM clientes_reciee WHERE nome ILIKE '%teste%';
