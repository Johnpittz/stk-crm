-- Testar exatamente o que o frontend faz
-- Rode no Supabase SQL Editor

-- 1. Buscar cliente RECIEE pelo CPF (como o frontend faz)
SELECT id, nome, cpf_cnpj 
FROM clientes_reciee 
WHERE cpf_cnpj = '70220987190'
LIMIT 1;

-- 2. Buscar faturas desse cliente
SELECT f.id, f.competencia, f.valor_total
FROM faturas_reciee f
JOIN clientes_reciee cr ON cr.id = f.cliente_id
WHERE cr.cpf_cnpj = '70220987190'
ORDER BY f.competencia DESC
LIMIT 12;

-- 3. Contar faturas
SELECT COUNT(*) as total_faturas
FROM faturas_reciee f
JOIN clientes_reciee cr ON cr.id = f.cliente_id
WHERE cr.cpf_cnpj = '70220987190';
