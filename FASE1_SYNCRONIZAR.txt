-- FASE 1 PASSO 2: Sincronizar clientes existentes
-- para a tabela unified_clients
-- Execute no SQL Editor do Supabase
-- ============================================

-- Sincronizar clientes do RECIEE
INSERT INTO unified_clients (source, source_id, nome, documento, uc, estado, distribuidora, grupo, created_at)
SELECT
  'reciee',
  id,
  nome,
  cpf_cnpj,
  uc,
  estado,
  distribuidora,
  grupo,
  created_at
FROM clientes_reciee
ON CONFLICT (source, source_id) DO UPDATE SET
  nome = EXCLUDED.nome,
  documento = EXCLUDED.documento,
  uc = EXCLUDED.uc,
  estado = EXCLUDED.estado,
  distribuidora = EXCLUDED.distribuidora,
  grupo = EXCLUDED.grupo;

-- Verificar resultado
SELECT source, COUNT(*) as total FROM unified_clients GROUP BY source;
