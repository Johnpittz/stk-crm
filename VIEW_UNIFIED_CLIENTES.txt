-- View que a página de clientes espera: v_unified_clientes
-- Mapeia unified_clients para o formato que a UI consome

CREATE OR REPLACE VIEW v_unified_clientes AS
SELECT
  id,
  nome,
  telefone,
  email,
  cpf_cnpj,
  cidade,
  estado,
  source AS origem,
  CASE WHEN EXISTS (
    SELECT 1 FROM atendimentos a
    WHERE a.cliente_id::text = unified_clients.id::text
       OR a.telefone_cliente = unified_clients.telefone
  ) THEN true ELSE false END AS tem_atendimento,
  CASE WHEN EXISTS (
    SELECT 1 FROM chatbot_sessions cs
    WHERE cs.telefone = unified_clients.telefone
  ) THEN true ELSE false END AS tem_chatbot,
  CASE WHEN EXISTS (
    SELECT 1 FROM faturas_reciee fr
    WHERE fr.cliente_id::text = unified_clients.id::text
  ) THEN true ELSE false END AS tem_faturas_reciee,
  created_at AS ultima_interacao
FROM unified_clients;
