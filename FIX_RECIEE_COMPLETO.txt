-- 1. Verificar quantos registros existem
SELECT 'clientes_reciee' as tabela, COUNT(*) as total FROM clientes_reciee
UNION ALL
SELECT 'unified_clients', COUNT(*) FROM unified_clients;

-- 2. Verificar sources no unified_clients
SELECT source, COUNT(*) as total FROM unified_clients GROUP BY source;

-- 3. Migrar clientes_reciee → unified_clients (se não migrou ainda)
INSERT INTO unified_clients (source, source_id, nome, documento, uc, estado, distribuidora, grupo, created_at)
SELECT
  'reciee',
  c.id::text,
  c.nome,
  c.cpf_cnpj,
  c.uc,
  c.estado,
  c.distribuidora,
  c.grupo,
  c.created_at
FROM clientes_reciee c
ON CONFLICT (source, source_id) DO NOTHING;

-- 4. Deletar view errada anterior
DROP VIEW IF EXISTS v_unified_clientes;

-- 5. Criar view correta (colunas que o CRM espera)
CREATE OR REPLACE VIEW v_unified_clientes AS
SELECT
  id,
  nome,
  telefone,
  email,
  documento AS cpf_cnpj,
  NULL AS cidade,
  estado,
  source AS origem,
  CASE WHEN source = 'cadastro' THEN true ELSE false END AS tem_atendimento,
  CASE WHEN source = 'chatbot' THEN true ELSE false END AS tem_chatbot,
  CASE WHEN source = 'reciee' THEN true ELSE false END AS tem_faturas_reciee,
  created_at AS ultima_interacao
FROM unified_clients;

-- 6. Conferir resultado final
SELECT COUNT(*) as total_view FROM v_unified_clientes;
SELECT origem, COUNT(*) as total FROM v_unified_clientes GROUP BY origem;
