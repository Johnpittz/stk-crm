-- ============================================
-- FASE 1: Tabela unificada de clientes
-- Execute este SQL no Supabase (SQL Editor)
-- ============================================

-- 1. Criar tabela
CREATE TABLE IF NOT EXISTS unified_clients (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  source TEXT NOT NULL,              -- 'cadastro', 'reciee', 'chatbot', 'disparo'
  source_id TEXT,                    -- ID na tabela de origem
  nome TEXT,
  documento TEXT,                    -- CPF ou CNPJ
  telefone TEXT,
  email TEXT,
  uc TEXT,                           -- Unidade Consumidora (RECIEE)
  estado TEXT,
  distribuidora TEXT,
  grupo TEXT,                        -- Grupo A/B (RECIEE)
  classificacao TEXT,                -- A/B/C/D (chatbot)
  status TEXT DEFAULT 'ativo',       -- ativo, inativo, lead
  metadata JSONB DEFAULT '{}'::jsonb, -- dados extras flexíveis
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- 2. Índices
CREATE INDEX IF NOT EXISTS idx_unified_clients_documento ON unified_clients(documento);
CREATE INDEX IF NOT EXISTS idx_unified_clients_telefone ON unified_clients(telefone);
CREATE INDEX IF NOT EXISTS idx_unified_clients_source ON unified_clients(source);

-- 3. RLS
ALTER TABLE unified_clients ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Service role unified_clients" ON unified_clients FOR ALL USING (true);
CREATE POLICY "Admin manage unified_clients" ON unified_clients
  FOR ALL USING (auth.uid() IN (SELECT id FROM profiles WHERE cargo = 'admin'));

-- 4. Migrar clientes do RECIEE
INSERT INTO unified_clients (source, source_id, nome, documento, uc, estado, distribuidora, grupo, created_at)
SELECT
  'reciee',
  id::text,
  nome,
  cpf_cnpj,
  uc,
  estado,
  distribuidora,
  grupo,
  created_at
FROM clientes_reciee
ON CONFLICT DO NOTHING;

-- 5. Migrar atendimentos (criar clientes a partir de conversas)
INSERT INTO unified_clients (source, source_id, nome, telefone, documento, created_at)
SELECT DISTINCT ON (telefone_cliente)
  'cadastro',
  id::text,
  nome_cliente,
  telefone_cliente,
  NULL,
  created_at
FROM atendimentos
WHERE telefone_cliente IS NOT NULL
  AND nome_cliente IS NOT NULL
  AND nome_cliente != 'Cliente'
ON CONFLICT DO NOTHING;

-- 6. Migrar chatbot sessions (leads qualificados)
INSERT INTO unified_clients (source, source_id, nome, telefone, classificacao, created_at)
SELECT DISTINCT ON (telefone)
  'chatbot',
  id::text,
  nome_lead,
  telefone,
  classificacao,
  created_at
FROM chatbot_sessions
WHERE telefone IS NOT NULL
  AND nome_lead IS NOT NULL
ON CONFLICT DO NOTHING;
